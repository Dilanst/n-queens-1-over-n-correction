  3 | \begin{lema}[Desplazamiento]\label{l:D}
  4 | En el marco de la Secci\'on~\ref{s:dual}, sean $y^0=(z^0,w^0)\in\Y$, $\tilde p:=p(y^0)$,
  5 | $J_0:=S(y^0)+\widehat\Pi$, $\varphi:=\|\F(y^0)\|_\infty$ y $\cM\ge\|J_0^{-1}\|_\infty$. Supongamos
  6 | $\tilde p_e\le1/22$ para toda casilla y pongamos
  7 | \[
  8 |  P^0:=\max_{\ell\in H\sqcup\Dg}\sum_{e\in\ell}\tilde p_e,\qquad h^0:=\max_fe^{w^0_f},\qquad
  9 |  \cL:=128\,e\,P^0+e^{1/8}h^0 .
 10 | \]
 11 | Si $2\cM\varphi\le1/8$ y $4\cM^2\cL\varphi\le1$, entonces:
 12 | \begin{enumerate}
 13 | \item[(D1)] $\F$ tiene un cero $y^*$ con $y^*-y^0\perp u$ y $\|y^*-y^0\|_\infty\le2\cM\varphi$; es el
 14 | \'unico
 15 | cero en $y^0+u^\perp$.
 16 | \item[(D2)] El paso de Newton $y^1:=y^0-J_0^{-1}\F(y^0)$ cumple $\|y^*-y^1\|_\infty\le2\cM^3\cL\varphi^2$.
 17 | \item[(D3)] $|\log(p_e(y^*)/\tilde p_e)|\le2d_e\|y^*-y^0\|_\infty\le16\cM\varphi$ y $d_ep_e(y^*)\le1/2$.
 18 | \end{enumerate}
 19 | \end{lema}
 20 | 
 21 | \begin{proof}
 22 | Sea $r:=2\cM\varphi\le1/8$ y $\mathbb B:=\{y:\|y-y^0\|_\infty\le r\}$.
 23 | 
 24 | \emph{Paso 0: $\mathbb B\subset\Y$, y en $\mathbb B$ se tiene $p_e\le e\,\tilde p_e$ y $d_ep_e<1/2$.} Sea
 25 | $y\in\mathbb B$, $y_t=y^0+t(y-y^0)$; $\theta_e(y_t)=\theta_e(y^0)+t\delta_e$ con $|\delta_e|\le d_er\le1/2$.
 26 | Mientras $d_ep_e\le1/2$ se tiene $d\log p_e/d\theta_e=(1-p_e)/(1-d_ep_e)\le2$. Sea $t^*$ el supremo de los
 27 | $t\le1$ tales que el segmento hasta $t$ est\'a en $\Y$ con $d_ep_e\le1/2$ para toda $e$. En $[0,t^*)$,
 28 | $\log(p_e(y_t)/\tilde p_e)\le2|\delta_e|\le1$, luego $d_ep_e\le4e/22<1/2$; por continuidad (si $\theta_e$
 29 | se acercara a $\bar\theta_{d_e}$, $p_e$ tender\'ia a $1/d_e$) $t^*=1$ y se alcanza.
 30 | 
 31 | \emph{Paso 1: $\|S(y)-S(y')\|_\infty\le\cL\|y-y'\|_\infty$ en $\mathbb B$.} La derivada de
 32 | $\tau_d(p_d(\theta))$ en $\theta$ es $\tau_d(1-2p+dp^2)/(1-dp)^2\in[0,8p]$ si $dp\le1/2$ (pues
 33 | $\tau_d\le2p$, el numerador es $\le1$ y el denominador $\ge1/4$), y $|\theta_e(y)-\theta_e(y')|\le
 34 | 4\|y-y'\|_\infty$. Por el paso 0, $|\tau_e(y)-\tau_e(y')|\le32\,e\,\tilde p_e\|y-y'\|_\infty$. En la fila
 35 | $\ell$ de $S(y)-S(y')$ cada casilla $e\in\ell$ aparece en $d_e\le4$ entradas, y la diagonal de una l\'inea
 36 | $f$ lleva adem\'as $e^{w_f}-e^{w'_f}$, de m\'odulo $\le e^{r}h^0|w_f-w'_f|$.
 37 | 
 38 | \emph{Paso 2: contracci\'on.} Sea $\Phi(y):=y-J_0^{-1}\F(y)$ en $\mathbb B':=\mathbb B\cap(y^0+u^\perp)$.
 39 | Como $J_0u=u$ y $J_0$ es sim\'etrica, $J_0^{-1}$ conserva $u^\perp$; como $\F\perp u$, $\Phi$ aplica
 40 | $y^0+u^\perp$ en s\'i mismo. Para $\xi\perp u$, $D\Phi(y)\xi=J_0^{-1}(S(y^0)-S(y))\xi$, luego
 41 | $\|D\Phi(y)\xi\|_\infty\le\cM\cL\|y-y^0\|_\infty\|\xi\|_\infty\le\frac12\|\xi\|_\infty$ en $\mathbb B'$
 42 | (porque $\cM\cL r=2\cM^2\cL\varphi\le\frac12$). Adem\'as $\|\Phi(y^0)-y^0\|_\infty\le\cM\varphi=r/2$. As\'i
 43 | $\Phi(\mathbb B')\subset\mathbb B'$ y $\Phi$ es una contracci\'on de raz\'on $\frac12$ en el convexo
 44 | $\mathbb B'$: tiene un \'unico punto fijo $y^*$, que es un cero de $\F$, con
 45 | $\|y^*-y^0\|_\infty\le\frac12\|y^*-y^0\|_\infty+\cM\varphi$. La unicidad en $y^0+u^\perp$ es el
 46 | Lema~\ref{l:P}(P2). Para (D2), $y^*-y^1=\Phi(y^*)-\Phi(y^0)=\int_0^1D\Phi(y_t)(y^*-y^0)\,dt$ con
 47 | $\|D\Phi(y_t)\xi\|_\infty\le\cM\cL\,t\,\|y^*-y^0\|_\infty\|\xi\|_\infty$, de donde
 48 | $\|y^*-y^1\|_\infty\le\frac12\cM\cL(2\cM\varphi)^2$. (D3) es el paso 0 con $y=y^*$.
 49 | \end{proof}

 51 | \begin{lema}[Punto de prueba desde torres]\label{l:PT}
 52 | Sea $z^0\in\R^H$ un punto dual de torres de la m\'ascara: $p^0_e:=p_2(a_e^\T z^0)$ tiene masas de fila y de
 53 | columna uno. Supongamos $d_ep^0_e\le1/2$ y $\ell^0_f:=\sum_{e\in f}p^0_e<1$, y sean
 54 | \[
 55 |  w^0_f:=\log(1-\ell^0_f),\qquad y^0:=(z^0,w^0),\qquad h_{\min}:=\min_f(1-\ell^0_f),
 56 | \]
 57 | \[
 58 |  \mu_e:=\sum_{f\ni e}(\ell^0_f-p^0_e),\qquad
 59 |  \beta_e:=\sum_{f\ni e}\log\frac{1-p^0_e}{1-\ell^0_f},\qquad
 60 |  \varphi_0:=\max_{\ell\in H\sqcup\Dg}\sum_{e\in\ell}p^0_e\mu_e,\qquad
 61 |  \varpi_2:=\tfrac12\sum_ep^0_e\mu_e .
 62 | \]
 63 | \textup{(}$\mu_e$ es la masa de las compa\~neras de $e$ en sus l\'ineas diagonales, y $\varpi_2$ la suma de
 64 | $p^0_ep^0_{e'}$ sobre los pares de casillas distintas de una misma l\'inea diagonal.\textup{)} Entonces
 65 | $y^0\in\Y$ y $\tilde p=p(y^0)$ cumple:
 66 | \begin{enumerate}
 67 | \item[(T1)] $\tilde p_e=p^0_e$ si $\Dg(e)=\emptyset$, y en general $g_{d_e}(\tilde
 68 | p_e)=g_{d_e}(p^0_e)-\beta_e$
 69 | con $0\le\beta_e\le\mu_e/h_{\min}$;
 70 | \item[(T2)] $0\le p^0_e-\tilde p_e\le2p^0_e\beta_e\le2p^0_e\mu_e/h_{\min}$;
 71 | \item[(T3)] $\F_\ell(y^0)=-\sum_{e\in\ell}(p^0_e-\tilde p_e)\le0$ para toda l\'inea $\ell\in H\sqcup\Dg$, de
 72 | modo que $\varphi=\|\F(y^0)\|_\infty\le2\varphi_0/h_{\min}$ y $\|\F(y^0)\|_1\le16\,\varpi_2/h_{\min}$.
 73 | \end{enumerate}
 74 | \end{lema}
 75 | 
 76 | \begin{proof}
 77 | $\theta_e(y^0)=g_2(p^0_e)+\sum_{f\ni e}\log(1-\ell^0_f)$ y
 78 | $g_{d_e}(p^0_e)=g_2(p^0_e)+|\Dg(e)|\log(1-p^0_e)$,
 79 | luego $\theta_e(y^0)=g_{d_e}(p^0_e)-\beta_e<\bar\theta_{d_e}$: $y^0\in\Y$ y (T1), con
 80 | $\beta_e=\sum_{f\ni e}\log\bigl(1+\frac{\ell^0_f-p^0_e}{1-\ell^0_f}\bigr)\le\mu_e/h_{\min}$. Como
 81 | $g_{d_e}$ es
 82 | creciente, $\tilde p_e\le p^0_e$; como $g_{d_e}'$ es decreciente, en $(0,p^0_e]$ vale al menos
 83 | $(1-d_ep^0_e)/(p^0_e(1-p^0_e))\ge1/(2p^0_e)$, y $g_{d_e}(p^0_e)-g_{d_e}(\tilde p_e)=\beta_e$ da (T2). Para
 84 | (T3), en $a\in H$ las masas de $p^0$ son uno, y en $f\in\Dg$, $e^{w^0_f}=1-\sum_{e\in f}p^0_e$. La norma
 85 | uno es $\sum_ed_e(p^0_e-\tilde p_e)\le4\cdot\frac2{h_{\min}}\sum_ep^0_e\mu_e$.
 86 | \end{proof}
