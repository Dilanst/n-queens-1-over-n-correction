  4 | \paragraph{El funcional de un lazo con la l\'inea.}
  5 | Para la estructura $B$ del Teorema~\ref{t:par} y $y\in\Y$ sean
  6 | \[
  7 |  J(y):=S(y)+\widehat\Pi=\sum_c\tau_c\,\iota_c\iota_c^\T+e^{w_f}\mathbf 1_f\mathbf 1_f^\T+\widehat\Pi,\qquad
  8 |  \Lambda(y):=\sum_c\ell_{d_c}(p_c(y))-\tfrac12\log\det J(y),
  9 | \]
 10 | con $\tau_c=\tau_{d_c}(p_c(y))$. Por la Observaci\'on~\ref{o:aplicacion} y el Lema~\ref{l:P},
 11 | $L(B)=\mathcal H(p^*)+\Lambda(y^*)+\frac12\log(4\pi k)$, de modo que
 12 | \begin{equation}\label{e:tres}
 13 |  \lambda(B)=\lambda_H+\bigl[\Lambda(y^0)-\Lambda_T(z^0)\bigr]+\bigl[\Lambda(y^*)-\Lambda(y^0)\bigr].
 14 | \end{equation}
 15 | 
 16 | \begin{lema}[Gradiente y hessiana de $\Lambda$]\label{l:Lam}
 17 | \begin{enumerate}
 18 | \item[(L1)] $\nabla\Lambda(y)=\sum_cg_c(y)\,\iota_c+g_w(y)\,\mathbf 1_f$, con
 19 | \[
 20 |  g_c(y):=\tau_c\bigl[\ell_{d_c}'(p_c)-\tfrac12\kappa_{d_c}(p_c)\,\widehat{\mathsf G}_y(c,c)\bigr],\qquad
 21 |  \widehat{\mathsf G}_y(c,c'):=\iota_c^\T J(y)^{-1}\iota_{c'},
 22 | \]
 23 | \[
 24 |  g_w(y):=-\tfrac12e^{w_f}\bigl[J(y)^{-1}\bigr]_{ff}.
 25 | \]
 26 | \item[(L2)] Para $k\ge k_0$ hay una constante $C_\Lambda$ tal que, en la bola $\mathbb B$ del
 27 | Lema~\ref{l:D}, $|\xi^\T\nabla^2\Lambda(y)\,\xi|\le C_\Lambda\|\xi\|_2^2$ para todo $\xi$, y
 28 | $\|\nabla\Lambda(y^0)\|_\infty\le C_\Lambda$.
 29 | \item[(L3)] $\bigl|\Lambda(y^*)-\Lambda(y^0)-\nabla\Lambda(y^0)^\T s\bigr|\le\frac12C_\Lambda\|\eta\|_2^2
 30 | +C_\Lambda\|\eta-s\|_1=O(k^{-4})$.
 31 | \end{enumerate}
 32 | \end{lema}
 33 | 
 34 | \begin{proof}
 35 | (L1) Derivando la composici\'on en una direcci\'on $\xi$, con $D
 36 | J[\xi]=\sum_c\tau_c\kappa_{d_c}(\iota_c^\T\xi)
 37 | \iota_c\iota_c^\T+e^{w_f}\xi_f\mathbf 1_f\mathbf 1_f^\T$:
 38 | $D\Lambda[\xi]=\sum_c\ell_{d_c}'\tau_c(\iota_c^\T\xi)-\frac12\tr(J^{-1}DJ[\xi])$.
 39 | 
 40 | (L2) Derivando otra vez,
 41 | \[
 42 |  D^2\Lambda[\xi,\xi]=\sum_c\varpi_c(\iota_c^\T\xi)^2-\tfrac12\tr\bigl(J^{-1}D^2J[\xi,\xi]\bigr)
 43 |  +\tfrac12\tr\bigl((J^{-1}DJ[\xi])^2\bigr),
 44 | \]
 45 | donde $\varpi_c$ es la derivada en $\theta$ de $\ell_d'(p_d)\tau_d(p_d)$ y $D^2J[\xi,\xi]=\sum_c\tau''_c
 46 | (\iota_c^\T\xi)^2\iota_c\iota_c^\T+e^{w_f}\xi_f^2\mathbf 1_f\mathbf 1_f^\T$, con $\tau''_c$ la derivada
 47 | segunda de $\tau_d(p_d)$. En $\mathbb B$ se tiene $d_cp_c\le\frac12$ y $p_c\le e\,\tilde p_c$, luego
 48 | $|\varpi_c|$, $|\tau''_c|$ y $\tau_c\kappa_{d_c}$ est\'an acotados por $c\,p_c$ (cada uno es $\tau_d$
 49 | por una
 50 | funci\'on acotada en $dp\le\frac12$). Por (D4), $\|J(y)^{-1}\|\le2e\,\cM_0$ y
 51 | $\widehat{\mathsf G}_y(c,c)\le3\|J(y)^{-1}\|$. Como en el Lema~\ref{l:D2},
 52 | $\sum_cp_c(\iota_c^\T\xi)^2\le4e\,P^0\|\xi\|_2^2$. Eso acota los dos primeros sumandos por $c\,\|\xi\|_2^2$.
 53 | Para el tercero, $\tr((J^{-1}DJ[\xi])^2)=\|J^{-1/2}DJ[\xi]J^{-1/2}\|_F^2\le\|J^{-1}\|^2\|DJ[\xi]\|_F^2$. Las
 54 | entradas de $DJ[\xi]$ son: en la diagonal, $\sum_{c\in\ell}\tau_c\kappa_{d_c}\iota_c^\T\xi$ (m\'as
 55 | $e^{w_f}\xi_f$ si $\ell=f$); fuera de ella, el t\'ermino de la \'unica casilla com\'un a las dos
 56 | l\'ineas, si existe (dos l\'ineas comparten a lo sumo una casilla: una m\'ascara tiene una casilla por
 57 | cruce, y $e$, $e'$ est\'an en filas y columnas distintas). Por Cauchy--Schwarz,
 58 | $(\sum_{c\in\ell}\tau_c\kappa_{d_c}\iota_c^\T\xi)^2\le c\sum_{c\in\ell}p_c(\iota_c^\T\xi)^2$; sumando en
 59 | $\ell$, y en los a lo sumo seis pares ordenados de l\'ineas de cada casilla,
 60 | $\|DJ[\xi]\|_F^2\le c\,\|\xi\|_2^2$. Por \'ultimo $|g_c(y^0)|\le c\,\tilde p_c$ y
 61 | $|g_w(y^0)|\le\frac12\cM_0$, lo que acota cada componente de $\nabla\Lambda(y^0)$.
 62 | 
 63 | (L3) El desarrollo de Taylor con resto integral en el segmento $[y^0,y^*]\subset\mathbb B$ da
 64 | $|\Lambda(y^*)-\Lambda(y^0)-\nabla\Lambda(y^0)^\T\eta|\le\frac12C_\Lambda\|\eta\|_2^2$, y
 65 | $|\nabla\Lambda(y^0)^\T(\eta-s)|\le\|\nabla\Lambda(y^0)\|_\infty\|\eta-s\|_1$. Se usa (P5).
 66 | \end{proof}
 67 | 
 68 | \begin{lema}[Valor en el punto de prueba]\label{l:I}
 69 | Para $k\ge k_0$, con $\gamma_e=\mathsf G(e,e)-2$ y $\gamma_{e'}=\mathsf G(e',e')-2$,
 70 | \[
 71 |  \Lambda(y^0)-\Lambda_T(z^0)=3pq+\tfrac{11}2(p^2q+pq^2)+\tfrac12(\gamma_e+\gamma_{e'})\,pq
 72 |  +pq\,\mathsf G(e,e')+O(k^{-4}).
 73 | \]
 74 | \end{lema}
 75 | 
 76 | \begin{proof}
 77 | Como $\tilde p_c=p^0_c$ fuera de $f$, por \eqref{e:bloque} y
 78 | $\det\tilde A=\det A^0\cdot\det(I+G^0(\tilde A-A^0))$,
 79 | \[
 80 |  \Lambda(y^0)-\Lambda_T(z^0)=\bigl[\ell_3(\tilde p_e)-\ell_2(p)\bigr]+\bigl[\ell_3(\tilde p_{e'})-\ell_2(q)
 81 |  \bigr]-\tfrac12\log\mathcal D-\tfrac12\log\tilde{\cS},
 82 | \]
 83 | exactamente, donde $\mathcal D$ es el determinante de orden dos
 84 | \[
 85 |  \mathcal D=(1+\Delta\tau_e\mathsf G(e,e))(1+\Delta\tau_{e'}\mathsf G(e',e'))
 86 |  -\Delta\tau_e\Delta\tau_{e'}\mathsf G(e,e')^2 .
 87 | \]
 88 | \emph{T\'ermino local.} $\ell_3(x)=\frac32x^2+4x^3+O(x^4)$, $\ell_2(x)=\frac12x^2+x^3+O(x^4)$ y
 89 | $\tilde p_e=p-pq-3p^2q+O(k^{-4})$ dan $\ell_3(\tilde p_e)-\ell_2(p)=p^2+3p^3-3p^2q+O(k^{-4})$. La suma
 90 | de los
 91 | dos es $p^2+q^2+3p^3+3q^3-3p^2q-3pq^2$.
 92 | 
 93 | \emph{Determinante.} Por (P2) y (E4), $\log\mathcal D=\Delta\tau_e\mathsf G(e,e)+\Delta\tau_{e'}\mathsf
 94 | G(e',e')+O(k^{-4})$, es decir,
 95 | \[
 96 |  \log\mathcal D=2p^2+2q^2-4pq+8p^3+8q^3-14(p^2q+pq^2)+\gamma_e(p^2-pq)+\gamma_{e'}(q^2-pq)+O(k^{-4}),
 97 | \]
 98 | y por (P4), $\log\tilde{\cS}=\tilde{\cS}-1+O(k^{-4})$. Sumando,
 99 | \[
100 |  \log\mathcal D+\log\tilde{\cS}=2p^2+2q^2-6pq+6p^3+6q^3-17(p^2q+pq^2)-(\gamma_e+\gamma_{e'})pq
101 |  -2pq\,\mathsf G(e,e')+O(k^{-4}).
102 | \]
103 | El t\'ermino local menos la mitad de esta suma es el enunciado.
104 | \end{proof}
