  3 | \begin{lema}[T\'ermino de primer orden]\label{l:II}
  4 | Para $k\ge k_0$,
  5 | $\ \nabla\Lambda(y^0)^\T s=\delta_e\,(\mathfrak c_e-\tfrac12)+\delta_{e'}\,(\mathfrak c_{e'}-\tfrac12)
  6 | +O(k^{-4})$.
  7 | \end{lema}

  9 | \begin{proof}
 10 | Por (P1), $s=J_0^{-1}(\delta_e\iota_e+\delta_{e'}\iota_{e'})$, luego $\nabla\Lambda(y^0)^\T s=\delta_e
 11 | \widehat{\mathfrak c}_e+\delta_{e'}\widehat{\mathfrak c}_{e'}$ con $\widehat{\mathfrak c}_c:=\iota_c^\T
 12 | J_0^{-1}\boldsymbol\Gamma$ y $\boldsymbol\Gamma:=\nabla\Lambda(y^0)=\boldsymbol\Gamma_H\oplus
 13 | \boldsymbol\Gamma_f$. Como $\delta_e=O(k^{-2})$, basta probar $\widehat{\mathfrak c}_e=\mathfrak c_e-\frac12
 14 | +O(k^{-2})$. Escribimos $g_c=g_c(y^0)$ y $\widehat{\mathsf G}=\widehat{\mathsf G}_{y^0}$.
 15 | 
 16 | (a) \emph{Casillas fuera de $f$.} Por \eqref{e:bloque}, $\widehat{\mathsf G}(c,c)=\tilde{\mathsf G}(c,c)+
 17 | (a_c^\T\tilde G\tilde v)^2/\tilde{\cS}$. Por (P3), (P4) y $|a_c^\T\tilde G\tilde v|\le8M_1(\tilde\tau_e+
 18 | \tilde\tau_{e'})$, $\widehat{\mathsf G}(c,c)=\mathsf G(c,c)+O(k^{-2})$ uniformemente en $c$, luego
 19 | $|g_c-g^0_c|\le c\,\tau^0_c/k^2$ y
 20 | \[
 21 |  \boldsymbol\Gamma_H=\Gamma^0+(g_e-g^0_e)\,a_e+(g_{e'}-g^0_{e'})\,a_{e'}+\varepsilon,\qquad
 22 |  \|\varepsilon\|_\infty=O(k^{-2}).
 23 | \]
 24 | 
 25 | (b) \emph{Las casillas del par.} Por \eqref{e:bloque} con $x=a_e$, $t=1$:
 26 | $\widehat{\mathsf G}(e,e)=\tilde{\mathsf G}(e,e)+(1-\tilde v^\T\tilde Ga_e)^2/\tilde{\cS}$, y
 27 | $\tilde v^\T\tilde Ga_e=\tilde\tau_e\tilde{\mathsf G}(e,e)+\tilde\tau_{e'}\tilde{\mathsf
 28 | G}(e',e)=2p+O(k^{-2})$.
 29 | Luego $\widehat{\mathsf G}(e,e)=3+O(1/k)$ y
 30 | $g_e=\tilde\tau_e[\ell_3'(\tilde p_e)-\frac12\kappa_3(\tilde p_e)\widehat{\mathsf
 31 | G}(e,e)]=-\frac32p+O(k^{-2})$.
 32 | Por (E5), $g^0_e=-p+O(k^{-2})$. As\'i $g_e-g^0_e=-\frac12p+O(k^{-2})$.
 33 | 
 34 | (c) \emph{La componente de la l\'inea.} $[J_0^{-1}]_{ff}=1/\tilde{\cS}$, luego $g_w=-\frac12\tilde h/
 35 | \tilde{\cS}=-\frac12+\frac12(p+q)+O(k^{-2})$ y
 36 | $\boldsymbol\Gamma_f=g_e+g_{e'}+g_w=-\frac12-(p+q)+O(k^{-2})$.
 37 | 
 38 | (d) Sea $\mathfrak a_e:=a_e^\T\tilde G\boldsymbol\Gamma_H$. Por (a), (P3) con $x=\Gamma^0$ y $x=\varepsilon$
 39 | ($\|\Gamma^0\|_\infty\le3$), (b) y (E4),
 40 | \[
 41 |  \mathfrak a_e=\mathfrak c_e+(g_e-g^0_e)\tilde{\mathsf G}(e,e)+(g_{e'}-g^0_{e'})\tilde{\mathsf G}(e,e')
 42 |  +O(k^{-2})=\mathfrak c_e-p+O(k^{-2}).
 43 | \]
 44 | 
 45 | (e) Por \eqref{e:bloque} con $x=\boldsymbol\Gamma_H$, $t=\boldsymbol\Gamma_f$:
 46 | $\sigma=(\boldsymbol\Gamma_f-\tilde\tau_e\mathfrak a_e-\tilde\tau_{e'}\mathfrak a_{e'})/\tilde{\cS}$. Como
 47 | $\mathfrak a_e=-1+O(1/k)$ por (E5), $\tilde\tau_e\mathfrak a_e=-p+O(k^{-2})$ y $\sigma=-\frac12+O(k^{-2})$.
 48 | 
 49 | (f) $\widehat{\mathfrak c}_e=a_e^\T(\tilde G\boldsymbol\Gamma_H-\sigma\tilde G\tilde v)+\sigma=
 50 | \mathfrak a_e+\sigma(1-\tilde v^\T\tilde Ga_e)=\mathfrak c_e-p-\frac12(1-2p)+O(k^{-2})=\mathfrak c_e-\frac12
 51 | +O(k^{-2})$.
 52 | \end{proof}

 54 | \begin{proof}[Prueba del Teorema~\ref{t:par}]
 55 | Por (P5), para $k\ge k_0$ existe el punto de Bethe de $B$ y se aplica el Corolario~\ref{c:par} con
 56 | $\cM=\cM_0$: $\lambda_H=-pq+O(k^{-4})$. Por (L3), el Lema~\ref{l:II}, (P1) y $\mathfrak c_e+1=O(1/k)$,
 57 | \[
 58 |  \Lambda(y^*)-\Lambda(y^0)=-\tfrac32(\delta_e+\delta_{e'})+pq\,[(\mathfrak c_e+1)+(\mathfrak c_{e'}+1)]
 59 |  +O(k^{-4}),
 60 | \]
 61 | y el segundo miembro es $-3pq-\frac92(p^2q+pq^2)+pq\,[(\mathfrak c_e+1)+(\mathfrak c_{e'}+1)]+O(k^{-4})$.
 62 | Con el Lema~\ref{l:I} y \eqref{e:tres},
 63 | \[
 64 |  \lambda(B)=-pq+pq\,\mathsf G(e,e')+pq\,(X_e+X_{e'})+O(k^{-4}),\qquad
 65 |  X_c:=p^0_c+\tfrac12(\mathsf G(c,c)-2)+(\mathfrak c_c+1).
 66 | \]
 67 | Por \'ultimo, $\tau^0_c/p^0_c=1+O(1/k)$, $\ell_2'(p)=p+O(k^{-2})$ y $\kappa_2(p)=1+2p+O(k^{-2})$ dan
 68 | $\ell_2'-\frac12\kappa_2\mathsf G(c,c)-\mathfrak c_c=-X_c+O(k^{-2})$, es decir, $\hat
 69 | c_c=-X_c+O(k^{-2})$. Las
 70 | cotas de $\mathsf G(e,e')$ y de $\hat c_e$ son (E4) y (E5).
 71 | \end{proof}
