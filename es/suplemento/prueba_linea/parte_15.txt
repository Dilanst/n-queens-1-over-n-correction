  4 | \begin{proof}[Paso 4 de la prueba del Lema~\ref{l:S}: $\tr((J_0^{-1}E)^2)=8\,p^2q^2+O_5$]
  5 | Como
  6 | $J_0^{-1}E=\sum_c\varepsilon_cJ_0^{-1}\iota_c\iota_c^\T+\varepsilon_wJ_0^{-1}\mathbf 1_f\mathbf 1_f^\T$,
  7 | \[
  8 |  \tr\bigl((J_0^{-1}E)^2\bigr)=\sum_{c,c'}\varepsilon_c\varepsilon_{c'}\,\widehat{\mathsf G}(c,c')^2
  9 |  +2\,\varepsilon_w\sum_c\varepsilon_c\,\bigl[(J_0^{-1}\iota_c)_f\bigr]^2
 10 |  +\frac{\varepsilon_w^2}{\tilde{\cS}^2}.
 11 | \]
 12 | \emph{Tama\~nos.} Por el paso 1 y (A3): $|\varepsilon_c|\le c\,p^0_c|\delta\theta_c|$, luego
 13 | $\sum_c|\varepsilon_c|=O_2$, $\sum_c\varepsilon_c^2=O_5$ y $\varepsilon_e,\varepsilon_{e'}=O_3$. Si
 14 | $a\in H_f$, las casillas de $a$ fuera de $f$ tienen $n_c=1$, salvo a lo sumo una con $n_c=2$ y peso
 15 | $O_1$; con $\kappa_c=1+O_1$, $m_a=1+O_1$ y $\tilde\tau_c=\tau^0_c$ fuera de $f$,
 16 | \[
 17 |  \sum_{c\in a,\,c\notin f}\varepsilon_c=pq\sum_{c\in a,\,c\notin f}\tau^0_c\kappa_c\,n_c+O_3=pq+O_3,
 18 |  \qquad \sum_{c\in a,\,c\notin f}|\varepsilon_c|=O_2 .
 19 | \]
 20 | Si $a\in H\setminus H_f$, a lo sumo dos casillas de $a$ tienen $n_c\ge1$, con peso $O_1$, y en las dem\'as
 21 | $\delta\theta_c=O_3$: $\sum_{c\in a}|\varepsilon_c|=O_3$. Por \'ultimo $\varepsilon_w=2pq+O_3$ por (A1).
 22 | 
 23 | \emph{El n\'ucleo.} Sean $c\ne c'$ fuera de $f$. Por \eqref{e:bloque},
 24 | $\widehat{\mathsf G}(c,c')=\tilde{\mathsf G}(c,c')+(a_c^\T\tilde G\tilde v)(a_{c'}^\T\tilde G\tilde v)/
 25 | \tilde{\cS}=\mathsf G(c,c')+O_2$ por (P3), y por (E3) y (E1) $\mathsf G(c,c')$ es el n\'umero de l\'ineas
 26 | comunes a $c$ y $c'$ (cero o una) m\'as $O_1$. Si $c\notin f$, $(J_0^{-1}\iota_c)_f=-\tilde v^\T\tilde G
 27 | a_c/\tilde{\cS}=O_1$. En todos los casos $|\widehat{\mathsf G}(c,c')|\le3\cM_0$ y
 28 | $|(J_0^{-1}\iota_c)_f|\le2\cM_0$.
 29 | 
 30 | \emph{La suma doble.} Los t\'erminos con $c\in f$ o $c'\in f$ suman $O_3\cdot O_2$; los de $c=c'$,
 31 | $O_5$; los de $c\ne c'$ sin l\'inea com\'un, $O_2\,(\sum_c|\varepsilon_c|)^2=O_6$. Quedan los pares
 32 | $c\ne c'$ fuera de $f$ con una l\'inea com\'un $a\in H$, con $\widehat{\mathsf G}(c,c')^2=1+O_1$:
 33 | \[
 34 |  \sum_{a\in H}\Bigl[\Bigl(\sum_{c\in a,\,c\notin f}\varepsilon_c\Bigr)^2
 35 |  -\sum_{c\in a,\,c\notin f}\varepsilon_c^2\Bigr]
 36 |  +O_1\sum_{a\in H}\Bigl(\sum_{c\in a,\,c\notin f}|\varepsilon_c|\Bigr)^2=4\,p^2q^2+O_5 ,
 37 | \]
 38 | porque las cuatro l\'ineas de $H_f$ dan $p^2q^2+O_5$ cada una y las otras $2k-4$ dan $O_6$ cada una.
 39 | 
 40 | \emph{Los otros dos t\'erminos.} $2\varepsilon_w\sum_c\varepsilon_c[(J_0^{-1}\iota_c)_f]^2=
 41 | O_2\,(O_2\cdot O_2+O_3)=O_5$ y $\varepsilon_w^2/\tilde{\cS}^2=4p^2q^2+O_5$. Con \eqref{e:S-traza}, eso
 42 | prueba el Lema~\ref{l:S}.
 43 | \end{proof}

 57 | \begin{lema}[Expresi\'on de $\widehat{\mathfrak c}_e$]\label{l:C}
 58 | Para $k\ge k_0$:
 59 | \begin{enumerate}
 60 | \item[(C1)] $(a_c^\T\tilde Ga_e,\ a_c^\T\tilde Ga_{e'})=\mathsf x_c^\T\mathbb W$ para toda casilla $c$, y
 61 | $\bigl(\tilde{\mathsf G}(c,c')\bigr)_{c,c'\in f}=\mathbb G\mathbb W$. En particular $\tilde v^\T\tilde
 62 | G\tilde v=\tilde\tau^\T\mathbb G\mathbb W\tilde\tau$ y $\tilde v^\T\tilde Ga_e=\epsilon_e^\T\mathbb G
 63 | \mathbb W\tilde\tau$: $\tilde{\cS}$, $\sigma_e$, $\mathsf n$, $\mathbb Q$ y $\mathsf m_e$ son funciones
 64 | de $p$, $q$ y $\mathbb G$.
 65 | \item[(C2)] Si $c\notin f$: $\widehat{\mathsf G}(c,c)-\mathsf G(c,c)=\mathsf x_c^\T\mathbb Q\,\mathsf x_c$
 66 | y $\widehat{\mathsf G}(e,c)=\mathsf x_c^\T\mathsf m_e$. Si $c\in f$:
 67 | $\widehat{\mathsf G}(e,c)=\mathsf x_c^\T\mathsf m_e+\sigma_e$.
 68 | \item[(C3)] $\widehat{\mathfrak c}_e=\mathsf m_e^\T\bigl[(\mathfrak c_e,\mathfrak c_{e'})^\T+\mathbb G\,
 69 | (g_e-g^0_e,\ g_{e'}-g^0_{e'})^\T-\tfrac12\mathfrak T\bigr]+\sigma_e\,(g_e+g_{e'}+g_w)$, y lo mismo con $e$
 70 | y $e'$ cambiadas.
 71 | \item[(C4)] $\mathfrak T=2\,(\mathbb Q_{ee},\mathbb Q_{e'e'})^\T+O_3$.
 72 | \end{enumerate}
 73 | \end{lema}
 74 | 
 75 | \begin{proof}
 76 | (C1) Sea $U=[a_e\ a_{e'}]$, de modo que $\tilde A=A^0+U\mathbb DU^\T$ y $U^\T G^0U=\mathbb G$. Entonces
 77 | $\tilde A\,G^0U\mathbb W=U\mathbb W+U\mathbb D\mathbb G\mathbb W=U$, es decir, $\tilde GU=G^0U\mathbb W$;
 78 | se multiplica por $a_c^\T$ y por $U^\T$. ($\mathbb W$ existe: $\mathbb D=O_2$ por (P2) y
 79 | $|\mathbb G_{cc'}|\le4M_1$.)
 80 | 
 81 | (C2) De $\tilde G=G^0-\tilde GU\mathbb DU^\T G^0$ y (C1), $\tilde{\mathsf G}(c,c)=\mathsf G(c,c)-
 82 | \mathsf x_c^\T\mathbb W\mathbb D\,\mathsf x_c$. Por el paso (a) de la prueba del Lema~\ref{l:II},
 83 | $\widehat{\mathsf G}(c,c)=\tilde{\mathsf G}(c,c)+(a_c^\T\tilde G\tilde v)^2/\tilde{\cS}$, y
 84 | $a_c^\T\tilde G\tilde v=\mathsf x_c^\T\mathbb W\tilde\tau=\mathsf x_c^\T\mathsf n$. Por \eqref{e:bloque}
 85 | con $x=a_e$, $t=1$: $J_0^{-1}\iota_e=(\tilde Ga_e-\sigma_e\tilde G\tilde v)\oplus\sigma_e$, luego
 86 | $\widehat{\mathsf G}(e,c)=\iota_c^\T J_0^{-1}\iota_e=a_c^\T\tilde G(a_e-\sigma_e\tilde v)+\sigma_e\mathbf
 87 | 1[c\in f]$, y $a_c^\T\tilde G(a_e-\sigma_e\tilde v)=\mathsf x_c^\T\mathbb W(\epsilon_e-\sigma_e\tilde\tau)$.
 88 | 
 89 | (C3) Por (L1), $\boldsymbol\Gamma=\sum_cg_c\iota_c+g_w\mathbf 1_f$, luego
 90 | $\widehat{\mathfrak c}_e=\sum_cg_c\,\widehat{\mathsf G}(e,c)+g_w\sigma_e$ y, por (C2),
 91 | $\widehat{\mathfrak c}_e=\mathsf m_e^\T\sum_cg_c\mathsf x_c+\sigma_e(g_e+g_{e'}+g_w)$. Si $c\notin f$,
 92 | $\tilde p_c=p^0_c$ y $g_c=g^0_c-\frac12\tau^0_c\kappa_2(p^0_c)(\widehat{\mathsf G}(c,c)-\mathsf G(c,c))$.
 93 | Como $\sum_cg^0_c\,\mathsf G(c,e)=a_e^\T G^0\Gamma^0=\mathfrak c_e$ (suma sobre todas las casillas), y
 94 | $\mathsf x_e=\mathbb G\epsilon_e$, $\mathsf x_{e'}=\mathbb G\epsilon_{e'}$,
 95 | \[
 96 |  \sum_cg_c\mathsf x_c=(\mathfrak c_e,\mathfrak c_{e'})^\T+(g_e-g^0_e)\,\mathsf x_e+(g_{e'}-g^0_{e'})\,
 97 |  \mathsf x_{e'}-\tfrac12\mathfrak T .
 98 | \]
 99 | 
100 | (C4) Las entradas de $\mathbb Q$ son $O_2$ ($\mathbb D=O_2$, $\mathsf n=O_1$, $\mathbb W$ acotada). Si
101 | $c\notin f$, por (E3) y (E1), $\mathsf G(c,e)$ es el n\'umero de l\'ineas comunes a $c$ y $e$ (cero o una)
102 | m\'as $O_1$, y $\tau^0_c\kappa_2(p^0_c)=\tau^0_c(1+O_1)$. Las casillas que comparten una l\'inea con $e$
103 | tienen peso total $m_{r(e)}+m_{c(e)}-2\tau^0_e=2+O_1$, y entre ellas a lo sumo dos comparten tambi\'en una
104 | l\'inea con $e'$, con peso $O_1$. En las casillas sin l\'inea com\'un con $e$ ni con $e'$,
105 | $|\mathsf x_c|=O_1$, y el peso total de las casillas es $\le3k$. Luego
106 | $\sum_{c\notin f}\tau^0_c\kappa_2(p^0_c)\,\mathsf G(c,e)^3=2+O_1$, lo mismo con $e'$, y las sumas de
107 | los productos c\'ubicos mixtos son $O_1$.
108 | \end{proof}

110 | \begin{proof}[Prueba del Teorema~\ref{t:par2}]
111 | Por \eqref{e:tres}, $\lambda=\lambda_H+[\Lambda(y^0)-\Lambda_T(z^0)]+[\Lambda(y^*)-\Lambda(y^0)]$.
112 | 
113 | \emph{Entrop\'ia.} Por la prueba de la Proposici\'on~\ref{p:H}, $\lambda_H=\mathfrak E_0-\mathfrak e$ con
114 | \[
115 |  \mathfrak E_0=\sum_{c\in f}\bigl[3\tilde p_c+2\log(1-\tilde p_c)-2p^0_c-\log(1-p^0_c)\bigr]
116 |  -(p+q)-\log(1-p-q),
117 | \]
118 | y por (T1) $\tilde p_e$ es la ra\'iz en $(0,\frac13)$ de $\tilde p_e(1-\tilde p_e)^2=p(1-p)(1-p-q)$:
119 | $\mathfrak E_0$ es una funci\'on de $(p,q)$. Por la identidad de la Observaci\'on~\ref{o:cubico},
120 | $\mathfrak e=\frac12\F(y^0)^\T J_0^{-1}\F(y^0)-\frac12\zeta^\T J_0\zeta-R_3$ con
121 | $|R_3|\le\frac16\cL\|\eta\|_\infty^2\|\eta\|_1=O_6$ por (A2), y $|\zeta^\T J_0\zeta|\le\|J_0\|_\infty
122 | \|\zeta\|_1^2=O_8$. Con (P1), $\mathfrak e=\frac12\sum_{c,c'\in f}\delta_c\delta_{c'}\widehat{\mathsf
123 | G}(c,c')+O_6$, y $\widehat{\mathsf G}(c,c')=(\mathbb G\mathbb W)_{cc'}+\sigma_c\sigma_{c'}\tilde{\cS}$
124 | en $f$.
125 | 
126 | \emph{Punto de prueba.} Por la prueba del Lema~\ref{l:I}, exactamente,
127 | $\Lambda(y^0)-\Lambda_T(z^0)=\sum_{c\in f}[\ell_3(\tilde p_c)-\ell_2(p^0_c)]-\frac12\log\det(I+\mathbb D
128 | \mathbb G)-\frac12\log\tilde{\cS}$.
129 | 
130 | \emph{Desplazamiento.} Por los Lemas~\ref{l:S} y~\ref{l:C}, $\Lambda(y^*)-\Lambda(y^0)=\delta_e
131 | \widehat{\mathfrak c}_e+\delta_{e'}\widehat{\mathfrak c}_{e'}+2p^2q^2+O_5$, y en (C3) se puede sustituir
132 | $\mathfrak T$ por $2(\mathbb Q_{ee},\mathbb Q_{e'e'})^\T$ con error $O_2\cdot O_3$.
133 | 
134 | \emph{Suma.} Sean $a=\mathsf G(e,e)-2$, $b=\mathsf G(e',e')-2$, $g=\mathsf G(e,e')$, $x=\mathfrak c_e+1$,
135 | $x'=\mathfrak c_{e'}+1$. Lo anterior da $\lambda(B)=\Phi(p,q,a,b,g,x,x')+O_5$, donde $\Phi$ es una
136 | funci\'on expl\'icita: una composici\'on de funciones racionales, logaritmos y la ra\'iz $\tilde p$, cuyos
137 | denominadores y argumentos de logaritmos ($1-dp$, $\det(I+\mathbb D\mathbb G)$, $\tilde{\cS}$, \dots) valen
138 | $1$ en el origen. Es anal\'itica en un entorno del origen de $\R^7$ que no depende de $k$, y sus siete
139 | argumentos son $O_1$ por (K), (E4) y (E5). Lo mismo vale para $\pi_1$, porque
140 | $\hat c_c=\frac{1-p^0_c}{1-2p^0_c}\bigl[\ell_2'(p^0_c)-\frac12\kappa_2(p^0_c)\mathsf G(c,c)-\mathfrak c_c
141 | \bigr]$. \textup{[O]} El polinomio de Taylor de grado cuatro de $\Phi+\pi_1$ en el origen, calculado con
142 | series truncadas de coeficientes racionales exactos (\texttt{orden4.py}), es
143 | $-\frac12p^2q^2+p^2q\,g+pq^2g$; los t\'erminos de grado $\le3$ se anulan, como dice el
144 | Teorema~\ref{t:par}. El resto de Taylor de una funci\'on anal\'itica da
145 | $|\Phi+\pi_1+\frac12p^2q^2-pq(p+q)g|\le c\,k^{-5}$.
146 | \end{proof}
